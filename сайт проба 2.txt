    // Загрузка/сохранение

    function saveGames() { localStorage.setItem('bfp_games', JSON.stringify(games)); }
    function savePosts() { localStorage.setItem('bfp_posts', JSON.stringify(posts)); }
    function saveDownloaded() { localStorage.setItem('bfp_downloaded', JSON.stringify(downloadedGames)); }

    // обновление счетчиков в профиле (сколько игр/постов от текущего пользователя Captain_Redbeard)
    function updateUserStats() {
        const currentUser = "Captain_Redbeard";
        const userGames = games.filter(g => g.author === currentUser).length;
        const userPosts = posts.filter(p => p.author === currentUser).length;
        const userGamesSpan = document.getElementById('userGamesCount');
        const userPostsSpan = document.getElementById('userPostsCount');
        if(userGamesSpan) userGamesSpan.innerText = userGames;
        if(userPostsSpan) userPostsSpan.innerText = userPosts;
    }

    // Отображение постов
    function renderPosts() {
        const container = document.getElementById('postsContainer');
        if(!container) return;
        if(posts.length === 0) {
            container.innerHTML = `<div class="post-card">Пока нет постов, будь первым! 🔥</div>`;
            return;
        }
        container.innerHTML = posts.map(post => `
            <div class="post-card">
                <div class="post-title"><i class="fas fa-quote-left"></i> ${escapeHtml(post.title)}</div>
                <div style="font-size:0.75rem; color:#c98686;">👤 ${escapeHtml(post.author)} | ${post.date}</div>
                <p style="margin-top: 0.7rem;">${escapeHtml(post.content)}</p>
                <div><i class="fas fa-comment-dots"></i> обсуждение в комментариях (0)</div>
            </div>
        `).join('');
    }

    // Генерация жанров из уникальных + рендер игр по фильтру
    let currentGenreFilter = "Все";
    function renderGenreButtons() {
        const genresSet = new Set(games.map(g => g.genre));
        const genreList = ["Все", ...Array.from(genresSet).sort()];
        const container = document.getElementById('genreFilterContainer');
        if(!container) return;
        container.innerHTML = genreList.map(genre => `<button class="genre-btn ${currentGenreFilter === genre ? 'active' : ''}" data-genre="${genre}">${genre}</button>`).join('');
        // добавить события
        document.querySelectorAll('.genre-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                currentGenreFilter = btn.getAttribute('data-genre');
                renderGamesByGenre();
                renderGenreButtons(); // обновить active
            });
        });
    }

    function renderGamesByGenre() {
        const container = document.getElementById('gamesListContainer');
        if(!container) return;
        let filtered = currentGenreFilter === "Все" ? games : games.filter(g => g.genre === currentGenreFilter);
        if(filtered.length === 0) {
            container.innerHTML = `<div class="game-card">⛔ Нет игр в этом жанре. Добавь первую через профиль!</div>`;
            return;
        }
        container.innerHTML = filtered.map(game => `
            <div class="game-card" data-game-id="${game.id}">
                <div class="game-title">🏴‍☠️ ${escapeHtml(game.name)}</div>
                <div class="game-genre"><i class="fas fa-tag"></i> ${escapeHtml(game.genre)}</div>
                <div class="game-desc">${escapeHtml(game.description)}</div>
                <div style="font-size:0.7rem; margin: 5px 0;">👤 ${escapeHtml(game.author)} | ⬇️ ${game.downloadsCount || 0}</div>
                <button class="btn-download download-game-btn" data-id="${game.id}" data-url="${game.apkUrl}"><i class="fas fa-download"></i> Скачать APK</button>
            </div>
        `).join('');
        // добавим обработчики на кнопки скачивания
        document.querySelectorAll('.download-game-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                const gameId = btn.getAttribute('data-id');
                const gameUrl = btn.getAttribute('data-url');
                const gameObj = games.find(g => g.id === gameId);
                if(gameObj) {
                    // увеличить счетчик скачиваний (эстетика)
                    gameObj.downloadsCount = (gameObj.downloadsCount || 0) + 1;
                    saveGames();
                    // добавить в "скачанные игры", если еще нет
                    if(!downloadedGames.includes(gameId)) {
                        downloadedGames.push(gameId);
                        saveDownloaded();
                        alert(`Игра "${gameObj.name}" добавлена в раздел "Скачанные игры"! Ссылка: ${gameUrl} (демо-ссылка)`);
                    } else {
                        alert(`Вы уже скачивали "${gameObj.name}". Ссылка для повторного скачивания: ${gameUrl}`);
                    }
                    renderGamesByGenre(); // обновить счетчик на карточке
                    renderDownloadedPage(); // обновить список скачанных
                    updateUserStats(); // (не обязательно)
                } else {
                    alert("Ошибка: игра не найдена");
                }
            });
        });
    }

    // отрисовка скачанных игр
    function renderDownloadedPage() {
        const container = document.getElementById('downloadedList');
        if(!container) return;
        const downloadedItems = downloadedGames.map(id => games.find(g => g.id === id)).filter(g => g);
        if(downloadedItems.length === 0) {
            container.innerHTML = `<li style="justify-content:center;">⚓ Сундук пуст. Скачай пару игр из раздела ИГРЫ!</li>`;
            return;
        }
        container.innerHTML = downloadedItems.map(game => `
            <li>
                <span><i class="fas fa-dice-d6"></i> <strong>${escapeHtml(game.name)}</strong> (${game.genre}) — ${game.description.substring(0, 50)}</span>
                <a href="${game.apkUrl}" target="_blank" style="background:#a52222; padding:0.3rem 0.8rem; border-radius: 20px; color:white; text-decoration:none; font-size:0.75rem;">Скачать снова</a>
            </li>
        `).join('');
    }

    // добавление новой игры (от Captain_Redbeard)
    function addNewGame(name, genre, description, apkUrl) {
        if(!name.trim() || !apkUrl.trim()) {
            alert("Название и ссылка APK обязательны!");
            return false;
        }
        const newId = 'g' + Date.now() + Math.random().toString(36).substr(2, 6);
        const newGame = {
            id: newId,
            name: name.trim(),
            genre: genre,
            description: description.trim() || "Без описания",
            apkUrl: apkUrl.trim(),
            author: "Captain_Redbeard",
            date: new Date().toISOString().slice(0,10),
            downloadsCount: 0
        };
        games.push(newGame);
        saveGames();
        alert(`Игра "${name}" добавлена в каталог!`);
        // обновить интерфейс
        renderGenreButtons();
        if(currentGenreFilter === "Все" || currentGenreFilter === genre) renderGamesByGenre();
        else renderGamesByGenre(); // все равно перерендер
        updateUserStats();
        return true;
    }

    function addNewPost(title, content) {
        if(!title.trim() || !content.trim()) {
            alert("Заголовок и текст поста обязательны!");
            return false;
        }
        const newPost = {
            id: 'p' + Date.now() + Math.random().toString(36).substr(2, 5),
            title: title.trim(),
            content: content.trim(),
            author: "Captain_Redbeard",
            date: new Date().toISOString().slice(0,10)
        };
        posts.unshift(newPost);
        savePosts();
        alert("Пост опубликован в ленте!");
        renderPosts();
        updateUserStats();
        return true;
    }

        // обновить активные ссылки
        document.querySelectorAll('.nav-links a').forEach(link => link.classList.remove('active'));
        const activeLink = Array.from(document.querySelectorAll('.nav-links a')).find(a => a.getAttribute('data-page') === pageId);
        if(activeLink) activeLink.classList.add('active');
        // дополнительный рендер при переключении
        if(pageId === 'posts') renderPosts();
        if(pageId === 'games') { renderGenreButtons(); renderGamesByGenre(); }
        if(pageId === 'downloaded') renderDownloadedPage();
        if(pageId === 'profile') updateUserStats();
    }

    // вспомогательная функция эскейпа
    function escapeHtml(str) {
        if(!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if(m === '&') return '&amp;';
            if(m === '<') return '&lt;';
            if(m === '>') return '&gt;';
            return m;
        }).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, function(c) {
            return c;
        });
    }
